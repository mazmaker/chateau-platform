-- สร้าง Tenant และกำหนดสิทธิ์ OWNER -- แก้ไข USER_ID ด้วย ID ที่ได้จากหน้า Users -- 1. สร้าง Tenant INSERT INTO tenants (name, slug) VALUES ('CHATEAU Platform', 'chateau-platform') ON CONFLICT (slug) DO NOTHING; -- 2. เชื่อม User กับ Tenant (แก้ USER_ID) INSERT INTO user_tenants (user_id, tenant_id, role, is_active, joined_at) SELECT 'USER_ID_HERE', id, 'owner', true, now() FROM tenants WHERE slug = 'chateau-platform' ON CONFLICT (user_id, tenant_id) DO UPDATE SET role = 'owner';

* แก้ USER_ID_HERE ด้วย User ID ที่ได้จากหน้า Users
