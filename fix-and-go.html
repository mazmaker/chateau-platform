🔧 Fix Database Error + Create User

Step 1: คัดลอก SQL นี้

DROP TRIGGER IF EXISTS on_auth_user_created ON auth.users; DROP FUNCTION IF EXISTS handle_new_user(); DROP TABLE IF EXISTS public.users CASCADE; CREATE TABLE public.users ( id uuid PRIMARY KEY, email text UNIQUE NOT NULL, full_name text, avatar_url text, preferences jsonb DEFAULT '{}', email_verified boolean DEFAULT false, is_active boolean DEFAULT true, created_at timestamptz DEFAULT now() ); CREATE OR REPLACE FUNCTION handle_new_user() RETURNS TRIGGER LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ BEGIN BEGIN INSERT INTO public.users (id, email, full_name) VALUES (NEW.id, NEW.email, COALESCE(NEW.raw_user_meta_data->>'full_name', split_part(NEW.email, '@', 1))); EXCEPTION WHEN OTHERS THEN NULL; END; RETURN NEW; END; $$; CREATE TRIGGER on_auth_user_created AFTER INSERT ON auth.users FOR EACH ROW EXECUTE FUNCTION handle_new_user();

คลิกที่ SQL ด้านบนเพื่อคัดลอก
